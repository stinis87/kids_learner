# The Ring event watcher (see ring_watcher.py) is enabled by default for every
# profile whenever Ring is configured; this file just tunes its timing here.
# Reachy polls each configured Ring device's event history every
# `poll_interval_seconds` and reacts — fetching a snapshot and speaking up —
# when a new motion or doorbell event appears, but at most once per device
# every `device_cooldown_seconds` to avoid nagging about repeated motion.
# Set `enabled=false` to opt this profile out entirely.
poll_interval_seconds=20
device_cooldown_seconds=120
