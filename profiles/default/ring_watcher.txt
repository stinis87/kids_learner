# The Ring event watcher (see ring_watcher.py) is enabled by default for every
# profile whenever Ring is configured; this file just tunes it here. Reachy
# subscribes to Ring's push notification stream (the same FCM mechanism the
# Ring app itself uses, not polling) and reacts — fetching a snapshot and
# speaking up — the moment a new motion or doorbell event arrives, but at most
# once per device every `device_cooldown_seconds` to avoid nagging about
# repeated motion (a doorbell "ding" is never held back by this cooldown).
# Set `enabled=false` to opt this profile out entirely.
device_cooldown_seconds=120

# door_call_mode controls how a doorbell "ding" is handled (motion events are
# unaffected and always just get a spoken nudge):
#   ask  (default) - Reachy asks out loud whether it should answer the door
#                    itself or let the person nearby talk, then opens a live
#                    two-way audio call accordingly using talk_to_door.
#   auto           - opens the live call immediately, no question asked, so
#                    Reachy answers the door fully on its own.
#   off            - back to a snapshot-only nudge, no call ever offered.
# A live call requires a device with a speaker/mic (most video doorbells/
# battery cams qualify) and the `aiortc` dependency.
door_call_mode=ask
